# 4. Compare Ratings Across Categories

# 5. Compare Ratings of Free and Paid Apps
plt.figure(figsize=(6,5))

plt.title("Ratings of Free vs Paid Apps")
plt.show()

# 6. Compare Average Installs Across Categories
avg_installs = (
    df.groupby("Category")["Installs"]
      .mean()
      .sort_values(ascending=False)
)

plt.figure(figsize=(12,6))
sns.barplot(
    x=avg_installs.index,
    y=avg_installs.values
)
plt.xticks(rotation=90)
plt.title("Average Installs by Category")
plt.xlabel("Category")
plt.ylabel("Average Installs")
plt.show()
