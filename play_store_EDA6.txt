# Task 1: Correlation Heatmap
# Select Numerical Columns
num_df = df[["Rating", "Reviews", "Installs", "Price", "Size"]]

# Correlation Matrix
corr = num_df.corr()

plt.figure(figsize=(8,6))
sns.heatmap(corr,
            annot=True,
            cmap="coolwarm",
            fmt=".2f",
            linewidths=0.5)

plt.title("Correlation Heatmap")
plt.show()
