# Section F – Feature Engineering (10 Marks)
# Task 1: Estimated Revenue
df["Estimated Revenue"] = df["Price"] * df["Installs"]

# Task 2: Create a Popularity Column
median_reviews = df["Reviews"].median()

df["Popularity"] = np.where(
    df["Reviews"] > median_reviews,
    "Popular",
    "Less Popular"
)

# Task 4: Create Install Groups
df["Install Group"] = pd.cut(
    df["Installs"],
    bins=[0, 10000, 100000, 1000000, float("inf")],
    labels=[
        "Below 10K",
        "10K - 100K",
        "100K - 1M",
        "Above 1M"
    ]
)

# 1. Which application category has the highest number of apps?
top_category = df["Category"].value_counts()
