# Section B – Data Cleaning & Preprocessing (20 Marks)

# Task 1
# Identify missing values in every column.

df.isnull().sum()

# Task 2: Remove Duplicate Records
df.duplicated().sum()

# Task 3: Handle Missing Values Using Suitable Techniques
# Fill Missing Director

df["director"]=df["director"].fillna("Unknown")

# Fill Missing Cast

df["cast"]=df["cast"].fillna("not available")

# Fill Missing Country
df["country"]=df["country"].fillna(df["country"].mode()[0])

# Fill Missing Rating
df["rating"]=df["rating"].fillna(df["rating"].mode()[0])

df = df.dropna(subset=["date_added"])

df["duration"]=df["duration"].fillna(df["duration"].mode()[0])

# Task 4: Convert the Date Added Column into DateTime Format
df["date_added"] = pd.to_datetime(df["date_added"], errors="coerce")

# Task 5: Extract New Columns
# Added Year
df["added_year"]=df["date_added"].dt.year

# Added Month
df["added_month"]=df["date_added"].dt.month

# Added Day

df["added_day"]=df["date_added"].dt.day
